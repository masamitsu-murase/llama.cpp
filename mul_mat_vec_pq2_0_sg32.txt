#version 450

#extension GL_EXT_shader_explicit_arithmetic_types_int32 : require

#include "mul_mat_vec_base.glsl"

layout(local_size_x_id = 0, local_size_y = 1, local_size_z = 1) in;

// PQ2_0 mat-vec for Ternary Bonsai 2 (q in {0,1,2}, q=3 never appears), Intel Xe1/Xe2 with subgroup size 32.
// Drop-in for mul_mat_vec.comp: same bindings, push constants, spec constants and reduce_result().
// BLOCK_SIZE must be a multiple of 32.
//
// One PQ2_0 block = 128 weights = 32 qs bytes -> one subgroup handles one block per iteration:
//   lane = qs byte index = 4 weights, ib is uniform inside the subgroup.
// Each B vec4 is loaded once and reused for all NUM_ROWS rows.
//
// dot(w, x) = d * sum((q - 1) * x) = d * (sum(q * x) - sum(x))
// sum(x) is shared by all rows.
// q is not shifted down: (byte & (3 << 2i)) = q_i * 4^i, and x_i is pre-scaled by 4^-i (exact in fp32).

#define PQ2_K       128u
#define PQ2_LANES   32u
#define PQ2_UNROLL  2u

const uvec4 PQ2_MASK  = uvec4(3u, 12u, 48u, 192u);
const vec4  PQ2_SCALE = vec4(1.0, 0.25, 0.0625, 0.015625);

uint a_blk0;    // block index of first_row, batch offset included
uint nblocks;   // blocks per row
uint b_offset;

void iter(inout FLOAT_TYPE temp[NUM_COLS][NUM_ROWS], const uint ib, const uint lane, const uint num_rows)
{
    uint  qb[NUM_ROWS];
    float dd[NUM_ROWS];

    [[unroll]] for (uint n = 0; n < num_rows; ++n) {
        const uint blk = a_blk0 + n * nblocks + ib;
        qb[n] = uint(data_a[blk].qs[lane]);
        dd[n] = float(data_a[blk].d);
    }

    [[unroll]] for (uint j = 0; j < NUM_COLS; ++j) {
        const vec4 x = vec4(data_b_v4[(j * p.batch_stride_b + b_offset) / 4 + ib * (PQ2_K / 4u) + lane]);
        const vec4 xs = x * PQ2_SCALE;
        const float xsum = (x.x + x.y) + (x.z + x.w);

        [[unroll]] for (uint n = 0; n < num_rows; ++n) {
            const float qdot = dot(vec4(uvec4(qb[n]) & PQ2_MASK), xs);
            temp[j][n] = fma(dd[n], qdot - xsum, temp[j][n]);
        }
    }
}

void compute_outputs(const uint first_row, const uint num_rows) {
    const uint tid  = gl_LocalInvocationID.x;
    const uint lane = tid % PQ2_LANES;
    const uint sub  = tid / PQ2_LANES;
    const uint step = BLOCK_SIZE / PQ2_LANES;

    uint a_offset, d_offset;
    get_offsets(a_offset, b_offset, d_offset);

    nblocks = p.ncols / PQ2_K;
    a_blk0  = a_offset + first_row * nblocks;

    FLOAT_TYPE temp[NUM_COLS][NUM_ROWS];

    [[unroll]] for (uint j = 0; j < NUM_COLS; ++j) {
        [[unroll]] for (uint n = 0; n < NUM_ROWS; ++n) {
            temp[j][n] = FLOAT_TYPE(0);
        }
    }

    uint ib = sub;

    while (ib + (PQ2_UNROLL - 1u) * step < nblocks) {
        [[unroll]] for (uint u = 0; u < PQ2_UNROLL; ++u) {
            iter(temp, ib + u * step, lane, num_rows);
        }
        ib += PQ2_UNROLL * step;
    }

    while (ib < nblocks) {
        iter(temp, ib, lane, num_rows);
        ib += step;
    }

    reduce_result(temp, d_offset, first_row, num_rows, tid);
}

void main() {
    const uint first_row = NUM_ROWS * (gl_WorkGroupID.x + gl_NumWorkGroups.x * gl_WorkGroupID.z);

    // do NUM_ROWS at a time, unless there aren't enough remaining rows
    if (first_row + NUM_ROWS <= p.stride_d) {
        compute_outputs(first_row, NUM_ROWS);
    } else {
        if (first_row >= p.stride_d) {
            return;
        }
        compute_outputs(first_row, min(NUM_ROWS, p.stride_d - first_row));
    }
}
